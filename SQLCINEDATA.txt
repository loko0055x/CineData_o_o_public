create table sala(
idsala int primary key ,
nombrepelicula varchar(40)not null,
imagenfoto image,
linktrailer varchar(max)not null,
);
go


--	TABLA ASIENTOS
create table asientos(
numasiento int ,
idsala int  ,
estado nchar(2),
constraint fksala foreign key (idsala) references sala (idsala) 
);
go

--	TABLA TIPOENTRADA
create table tipoentrada(
id int primary key,
tipoentrada varchar(30),
precio money,
);
go

--	TABLA DULCERIA
create table dulceria(
iddulceria int primary key,
nombre varchar(30),
precio money,
);
go

--	TABLA ADMINISTRADORES
create table administradores(
dni int primary key,
nombre varchar (30) not null,
apellido varchar (40)not null,
salario money not null,
correo varchar (max)not null,
contraseña  varchar (max)not null,
dirrecion varchar (100)null,
telefono bigint null,
sexo char (1)not null,
fechanacimiento date null,
);

--	TABLA BOLETA


create table boleta (
idboleta int identity primary key,
nombre varchar(30)not null,
apellido varchar(30)not null,
correo varchar(100)not null,
nombrepelicula varchar (30)not null ,--fk
subtotalentradas money not null,
subtotaldulceria money not null,
fecha_compra datetime default getdate(),
tipo_pago varchar(30)not null,
total money not null, 
);


go

--	TABLA HISTORIAL_CLIENTE_PREMIUM
create table historialclientepremiun (
codhistorial int primary key identity,
dni bigint not null,
fecha datetime default getdate(),
notalogin varchar (max)not null,
);



--	TABLA EMPLEADOS
create table empleados(
idempleado nchar(5)primary key,
nombre varchar (30) not null,
apellido varchar (40)not null,
salario money not null,
correo varchar (max) null, 
dirrecion varchar (100)null, 
telefono bigint null,
sexo char (1) null,
cargo nvarchar (20),
fechanacimiento date null,
horasalasemana varchar(10) not null,
);

--	TABLA CATEGORIA_PELI
create table categoriapeli(
codipeli int primary key ,
nombrecategoria varchar(25),
censura varchar (20),
formato varchar(10),
idioma varchar(25),
);
